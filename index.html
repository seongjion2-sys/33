import base64 import glob files = glob.glob('*.png') + glob.glob('*.jpg') + glob.glob('*.jpeg') print("Found files:", files) img_path = files[0] if files else None if img_path: with open(img_path, 'rb') as f: img_data = f.read() b64_str = base64.b64encode(img_data).decode('utf-8') data_url = f"data:image/png;base64,{b64_str}" else: data_url = "" html_template = """ 인간 키우기
나이: 1세 기분: 우는 중
나이: 1세
🖐️ 쓰다듬기
레벨: 0
🍼 밥 먹이기
레벨: 0
🛈 산책시키기
레벨: 0
""" final_html = html_template.replace('__DATA_URL__', data_url) with open('index.html', 'w', encoding='utf-8') as f: f.write(final_html) print("Generated index.html with size:", len(final_html))
